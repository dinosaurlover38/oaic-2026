\iftoggle{ANIMATIONS}{% True
    \newenvironment{noanimationframe}[2][\unskip]{
        \settoggle{ANIMATIONS}{false}
        % Renew existing animation commands
        \ifthenelse{\equal{#1}{\unskip}}{
            \begin{frame}{#2}
        }{
            \begin{frame}[#1]{#2}
        }
    }{
        \end{frame}
        \settoggle{ANIMATIONS}{true}
    }
}{% False
    \newenvironment{noanimationframe}[2][\unskip]{

            }{

            }
